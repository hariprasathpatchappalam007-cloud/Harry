SELECT
    LEAD_NO,
    COUNT(*) AS CO_APPLICANT_COUNT
FROM LEAD_DETAILS_CUSTOMER
WHERE CUSTTYPE = 'co-applicant'
GROUP BY LEAD_NO
HAVING COUNT(*) > 2
ORDER BY CO_APPLICANT_COUNT DESC;

SELECT *
FROM LEAD_DETAILS_CUSTOMER
WHERE CUSTTYPE = 'co-applicant'
AND LEAD_NO IN
(
    SELECT LEAD_NO
    FROM LEAD_DETAILS_CUSTOMER
    WHERE CUSTTYPE = 'co-applicant'
    GROUP BY LEAD_NO
    HAVING COUNT(*) > 2
);
