SELECT
    STRING_AGG(
        CAST(
            COLUMN_NAME + ' ' +
            DATA_TYPE +
            CASE
                WHEN CHARACTER_MAXIMUM_LENGTH IS NOT NULL
                THEN '(' +
                    CASE
                        WHEN CHARACTER_MAXIMUM_LENGTH = -1 THEN 'MAX'
                        ELSE CAST(CHARACTER_MAXIMUM_LENGTH AS VARCHAR(10))
                    END + ')'
                WHEN DATA_TYPE IN ('decimal', 'numeric')
                THEN '(' + CAST(NUMERIC_PRECISION AS VARCHAR(10)) +
                     ',' + CAST(NUMERIC_SCALE AS VARCHAR(10)) + ')'
                ELSE ''
            END +
            CASE WHEN IS_NULLABLE = 'NO' THEN ' NOT NULL' ELSE ' NULL' END
        AS VARCHAR(MAX)),
        ', '
    ) WITHIN GROUP (ORDER BY ORDINAL_POSITION) AS TableStructure
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_NAME = 'Core_flex_all_apps_master';
