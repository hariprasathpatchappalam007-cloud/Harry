USE DFM;

SELECT
    @@SERVERNAME AS ConnectedServer,
    DB_NAME() AS ConnectedDatabase,
    ORIGINAL_LOGIN() AS ConnectedLogin,
    OBJECT_ID(N'dbo.sp_SaveBudgetLine', N'P') AS ProcedureId,
    HAS_PERMS_BY_NAME(
        N'dbo.sp_SaveBudgetLine', N'OBJECT', N'VIEW DEFINITION'
    ) AS CanViewDefinition,
    OBJECT_DEFINITION(
        OBJECT_ID(N'dbo.sp_SaveBudgetLine', N'P')
    ) AS ProcedureDefinition;
