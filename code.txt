Get-ChildItem "$env:USERPROFILE\Downloads" -Filter "*CoworkerCertificate*" |
    Select-Object Name, FullName

$script = Join-Path $env:USERPROFILE 'Downloads\Test-CoworkerCertificate.ps1'
$report = Join-Path $env:USERPROFILE 'Downloads\Coworker-Certificate-Report.txt'

if (-not (Test-Path -LiteralPath $script -PathType Leaf)) {
    throw "Script has not been copied to this PC: $script"
}

& $script | Tee-Object -FilePath $report
