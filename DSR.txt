/*
    Returns the latest SOA_CallLog_SCORECARD row per lead, in batches.

    lead_details and lead_details_finance are read from the current database.
    Scorecard lookup order:
      1. Latest SAS_ScoreCard method row in tamweel_old
      2. Latest SAS_ScoreCard method row in tamweel_archival, only when no
         matching row exists in tamweel_old

    Edit @FromUpdatedOnDate and @ToUpdatedOnDate below to set the hard-coded
    lead_details date range. The To date is inclusive; either date may be set to
    NULL to leave that boundary open.
    Filtered lead/finance rows are staged once. Each distinct lead is looked up
    once in tamweel_old; only missing leads are looked up in tamweel_archival.
    @BatchSize can be adjusted to suit the server workload.
    For best performance, ensure both scorecard tables have an index beginning
    with (lead_no, method_name, createdon_date DESC), with RequestXML included
    if practical.
    Also check indexes on lead_details (lead_stage, updatedon_date, lead_no)
    INCLUDE (app_ref_no), and lead_details_finance (lead_no) INCLUDE (LEAD_DSR).
    Review existing indexes and the actual execution plan before adding indexes.
*/

SET NOCOUNT ON;

DECLARE @BatchSize int = 500;
DECLARE @FromUpdatedOnDate date = '2026-01-01';
DECLARE @ToUpdatedOnDate date = '2026-11-12';
DECLARE @BatchStart bigint = 1;
DECLARE @BatchEnd bigint;
DECLARE @MaxBatchRow bigint;

IF @FromUpdatedOnDate IS NOT NULL
   AND @ToUpdatedOnDate IS NOT NULL
   AND @FromUpdatedOnDate > @ToUpdatedOnDate
BEGIN
    THROW 50001, 'FromUpdatedOnDate must be on or before ToUpdatedOnDate.', 1;
END;

IF OBJECT_ID('tempdb..#LeadBatch') IS NOT NULL DROP TABLE #LeadBatch;
IF OBJECT_ID('tempdb..#CurrentBatch') IS NOT NULL DROP TABLE #CurrentBatch;
IF OBJECT_ID('tempdb..#FilteredLeads') IS NOT NULL DROP TABLE #FilteredLeads;
IF OBJECT_ID('tempdb..#CurrentScorecards') IS NOT NULL DROP TABLE #CurrentScorecards;
IF OBJECT_ID('tempdb..#DSRResult') IS NOT NULL DROP TABLE #DSRResult;

SELECT
    ld.lead_stage,
    ld.lead_no,
    ld.app_ref_no,
    ld.updatedon_date AS ld_updateon,
    ldf.LEAD_DSR AS Fin_DSR
INTO #FilteredLeads
FROM lead_details ld
INNER JOIN lead_details_finance ldf
    ON ldf.lead_no = ld.lead_no
WHERE ld.lead_no IS NOT NULL
  AND ld.lead_stage IN ('CUST_OL_ACCEPT', 'OPS_PMU', 'OPS_LODGMENT', 'End_Of_Process')
  AND (@FromUpdatedOnDate IS NULL OR ld.updatedon_date >= @FromUpdatedOnDate)
  AND (@ToUpdatedOnDate IS NULL OR ld.updatedon_date < DATEADD(day, 1, @ToUpdatedOnDate))
OPTION (RECOMPILE);

CREATE CLUSTERED INDEX IX_FilteredLeads_LeadNo
    ON #FilteredLeads (lead_no);

SELECT
    ROW_NUMBER() OVER (ORDER BY leads.lead_no) AS BatchRow,
    leads.lead_no
INTO #LeadBatch
FROM
(
    SELECT DISTINCT lead_no
    FROM #FilteredLeads
) AS leads;

CREATE UNIQUE CLUSTERED INDEX IX_LeadBatch_BatchRow
    ON #LeadBatch (BatchRow);

SELECT TOP (0)
    batch.lead_no,
    SCC.createdon_date AS SCC_createdon_date,
    CAST(SCC.RequestXML AS XML).value(
        '(//*[local-name()="Dsr"]/text())[1]',
        'DECIMAL(10,2)'
    ) AS SCC_DSR
INTO #CurrentScorecards
FROM #LeadBatch batch
LEFT JOIN [tamweel_old].[dbo].[SOA_CallLog_SCORECARD] SCC
    ON 1 = 0;

CREATE UNIQUE CLUSTERED INDEX IX_CurrentScorecards_LeadNo
    ON #CurrentScorecards (lead_no);

SELECT TOP (0)
    leads.lead_stage,
    leads.lead_no,
    leads.app_ref_no,
    leads.ld_updateon,
    SCC.SCC_createdon_date,
    SCC.SCC_DSR,
    leads.Fin_DSR
INTO #DSRResult
FROM #FilteredLeads leads
CROSS JOIN #CurrentScorecards SCC;

SELECT TOP (0) lead_no
INTO #CurrentBatch
FROM #LeadBatch;

CREATE UNIQUE CLUSTERED INDEX IX_CurrentBatch_LeadNo
    ON #CurrentBatch (lead_no);

SELECT @MaxBatchRow = MAX(BatchRow)
FROM #LeadBatch;

WHILE @BatchStart <= ISNULL(@MaxBatchRow, 0)
BEGIN
    SET @BatchEnd = @BatchStart + @BatchSize - 1;

    TRUNCATE TABLE #CurrentBatch;
    TRUNCATE TABLE #CurrentScorecards;

    INSERT INTO #CurrentBatch (lead_no)
    SELECT lead_no
    FROM #LeadBatch
    WHERE BatchRow BETWEEN @BatchStart AND @BatchEnd;

    -- First use the newest scorecard row found in tamweel_old.
    INSERT INTO #CurrentScorecards
    (
        lead_no,
        SCC_createdon_date,
        SCC_DSR
    )
    SELECT
        batch.lead_no,
        SCC.createdon_date AS SCC_createdon_date,
        CAST(SCC.RequestXML AS XML).value(
            '(//*[local-name()="Dsr"]/text())[1]',
            'DECIMAL(10,2)'
        ) AS SCC_DSR
    FROM #CurrentBatch batch
    CROSS APPLY
    (
        SELECT TOP (1)
            old_scc.createdon_date,
            old_scc.RequestXML
        FROM [tamweel_old].[dbo].[SOA_CallLog_SCORECARD] old_scc
        WHERE old_scc.lead_no = batch.lead_no
          AND old_scc.method_name = 'SAS_ScoreCard'
        ORDER BY old_scc.createdon_date DESC
    ) AS SCC;

    -- Only leads with no scorecard row in tamweel_old are looked up in archival.
    INSERT INTO #CurrentScorecards
    (
        lead_no,
        SCC_createdon_date,
        SCC_DSR
    )
    SELECT
        batch.lead_no,
        SCC.createdon_date AS SCC_createdon_date,
        CAST(SCC.RequestXML AS XML).value(
            '(//*[local-name()="Dsr"]/text())[1]',
            'DECIMAL(10,2)'
        ) AS SCC_DSR
    FROM #CurrentBatch batch
    CROSS APPLY
    (
        SELECT TOP (1)
            archive_scc.createdon_date,
            archive_scc.RequestXML
        FROM [tamweel_archival].[dbo].[SOA_CallLog_SCORECARD] archive_scc
        WHERE archive_scc.lead_no = batch.lead_no
          AND archive_scc.method_name = 'SAS_ScoreCard'
        ORDER BY archive_scc.createdon_date DESC
    ) AS SCC
    WHERE NOT EXISTS
    (
        SELECT 1
        FROM #CurrentScorecards existing
        WHERE existing.lead_no = batch.lead_no
    );

    INSERT INTO #DSRResult
    (
        lead_stage,
        lead_no,
        app_ref_no,
        ld_updateon,
        SCC_createdon_date,
        SCC_DSR,
        Fin_DSR
    )
    SELECT
        leads.lead_stage,
        leads.lead_no,
        leads.app_ref_no,
        leads.ld_updateon,
        SCC.SCC_createdon_date,
        SCC.SCC_DSR,
        leads.Fin_DSR
    FROM #CurrentScorecards SCC
    INNER JOIN #FilteredLeads leads
        ON leads.lead_no = SCC.lead_no;

    SET @BatchStart = @BatchEnd + 1;
END;

SELECT
    lead_stage,
    lead_no,
    app_ref_no,
    ld_updateon,
    SCC_createdon_date,
    SCC_DSR,
    Fin_DSR,
    SCC_DSR - Fin_DSR AS DSR_Difference
FROM #DSRResult
ORDER BY ld_updateon DESC;

DROP TABLE #CurrentBatch;
DROP TABLE #CurrentScorecards;
DROP TABLE #DSRResult;
DROP TABLE #LeadBatch;
DROP TABLE #FilteredLeads;
