/*
    Read-only monthly DSR difference report.

    Iterates one Createdon_date month at a time from January 2018 through
    October 2025,     in one-month date windows. Returns only non-zero, calculable
    DSR differences, with month and lead identifiers so the results are
    traceable. It does not update finance.

    Scorecard priority: newest SAS_ScoreCard row in tamweel_old; use
    tamweel_archival only when no matching old-database row exists.
    Existing stages and newest-first lead update ordering are preserved.

    For best performance, review existing indexes and the execution plan:
      lead_details (lead_stage, Createdon_date, lead_no) INCLUDE (app_ref_no)
      lead_details_finance (lead_no) INCLUDE (LEAD_DSR)
      scorecard tables (lead_no, method_name, createdon_date DESC),
        optionally INCLUDE (RequestXML), subject to its type and size.
*/

SET NOCOUNT ON;

DECLARE @FirstMonth date = '20180101';
DECLARE @AfterLastMonth date = '20251101';
DECLARE @MonthStart date = @FirstMonth;
DECLARE @MonthEnd date;
DECLARE @MonthYear int;
DECLARE @MonthNumber int;

WHILE @MonthStart < @AfterLastMonth
BEGIN
    SET @MonthEnd = DATEADD(MONTH, 1, @MonthStart);

    IF OBJECT_ID('tempdb..#MonthLeads') IS NOT NULL DROP TABLE #MonthLeads;
    IF OBJECT_ID('tempdb..#MonthLeadKeys') IS NOT NULL DROP TABLE #MonthLeadKeys;
    IF OBJECT_ID('tempdb..#MonthScorecards') IS NOT NULL DROP TABLE #MonthScorecards;

    SELECT DISTINCT
        ld.lead_stage,
        ld.lead_no,
        ld.app_ref_no,
        ld.updatedon_date AS ld_updateon,
        ldf.LEAD_DSR AS Fin_DSR
    INTO #MonthLeads
    FROM lead_details ld
    INNER JOIN lead_details_finance ldf
        ON ldf.lead_no = ld.lead_no
    WHERE ld.lead_no IS NOT NULL
      AND ld.lead_stage IN ('CUST_OL_ACCEPT', 'OPS_PMU', 'OPS_LODGMENT', 'End_Of_Process')
      AND ld.Createdon_date >= @MonthStart
      AND ld.Createdon_date < @MonthEnd
    OPTION (RECOMPILE);

    CREATE CLUSTERED INDEX IX_MonthLeads_LeadNo
        ON #MonthLeads (lead_no);

    SELECT
        DISTINCT lead_no
    INTO #MonthLeadKeys
    FROM #MonthLeads;

    CREATE UNIQUE CLUSTERED INDEX IX_MonthLeadKeys_LeadNo
        ON #MonthLeadKeys (lead_no);

    SELECT TOP (0)
        leads.lead_no,
        SCC.createdon_date AS SCC_createdon_date,
        CAST(SCC.RequestXML AS XML).value(
            '(//*[local-name()="Dsr"]/text())[1]',
            'DECIMAL(38,20)'
        ) AS SCC_DSR
    INTO #MonthScorecards
    FROM #MonthLeadKeys leads
    LEFT JOIN [tamweel_old].[dbo].[SOA_CallLog_SCORECARD] SCC
        ON 1 = 0;

    CREATE UNIQUE CLUSTERED INDEX IX_MonthScorecards_LeadNo
        ON #MonthScorecards (lead_no);

    -- The old database wins whenever it has any matching method row.
    INSERT INTO #MonthScorecards (lead_no, SCC_createdon_date, SCC_DSR)
    SELECT
        leads.lead_no,
        SCC.createdon_date,
        CAST(SCC.RequestXML AS XML).value(
            '(//*[local-name()="Dsr"]/text())[1]',
            'DECIMAL(38,20)'
        )
    FROM #MonthLeadKeys leads
    CROSS APPLY
    (
        SELECT TOP (1) old_scc.createdon_date, old_scc.RequestXML
        FROM [tamweel_old].[dbo].[SOA_CallLog_SCORECARD] old_scc
        WHERE old_scc.lead_no = leads.lead_no
          AND old_scc.method_name = 'SAS_ScoreCard'
        ORDER BY old_scc.createdon_date DESC
    ) AS SCC;

    -- Query archival only for leads not found in tamweel_old.
    INSERT INTO #MonthScorecards (lead_no, SCC_createdon_date, SCC_DSR)
    SELECT
        leads.lead_no,
        SCC.createdon_date,
        CAST(SCC.RequestXML AS XML).value(
            '(//*[local-name()="Dsr"]/text())[1]',
            'DECIMAL(38,20)'
        )
    FROM #MonthLeadKeys leads
    CROSS APPLY
    (
        SELECT TOP (1) archive_scc.createdon_date, archive_scc.RequestXML
        FROM [tamweel_archival].[dbo].[SOA_CallLog_SCORECARD] archive_scc
        WHERE archive_scc.lead_no = leads.lead_no
          AND archive_scc.method_name = 'SAS_ScoreCard'
        ORDER BY archive_scc.createdon_date DESC
    ) AS SCC
    WHERE NOT EXISTS
    (
        SELECT 1
        FROM #MonthScorecards existing
        WHERE existing.lead_no = leads.lead_no
    );

    SET @MonthYear = YEAR(@MonthStart);
    SET @MonthNumber = MONTH(@MonthStart);
    RAISERROR(N'Completed DSR comparison for %04d-%02d. Returning this month''s differences.',
        0, 1, @MonthYear, @MonthNumber) WITH NOWAIT;

    SELECT
        @MonthStart AS ReportMonth,
        leads.lead_stage,
        leads.lead_no,
        leads.app_ref_no,
        scorecards.SCC_DSR - leads.Fin_DSR AS DSR_Difference
    FROM #MonthLeads leads
    INNER JOIN #MonthScorecards scorecards
        ON scorecards.lead_no = leads.lead_no
    WHERE scorecards.SCC_DSR IS NOT NULL
      AND leads.Fin_DSR IS NOT NULL
      AND scorecards.SCC_DSR <> leads.Fin_DSR
    ORDER BY leads.ld_updateon DESC, leads.lead_no;

    DROP TABLE #MonthScorecards;
    DROP TABLE #MonthLeadKeys;
    DROP TABLE #MonthLeads;

    SET @MonthStart = @MonthEnd;
END;
