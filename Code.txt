using System;
using System.Globalization;
using System.Text;
using System.Text.RegularExpressions;
using System.Web.UI;

public partial class CustomerScript : Page
{
    protected void btnGenerate_Click(object sender, EventArgs e)
    {
        lblMessage.Text = "";
        txtSQL.Text = "";

        string customerName = txtCustomerName.Text.Trim();
        string cif = txtCIF.Text.Trim();
        string emiratesID = txtEmiratesID.Text.Trim();
        string dobInput = txtDOB.Text.Trim();
        string passport = txtPassport.Text.Trim();
        string nationality = txtNationality.Text.Trim();
        string mobile = txtMobile.Text.Trim();

        // Validate mandatory fields.
        if (customerName == "" || cif == "" ||
            emiratesID == "" || dobInput == "" ||
            passport == "" || nationality == "" ||
            mobile == "")
        {
            lblMessage.Text = "Please enter all customer details.";
            return;
        }

        // Validate CIF.
        if (!Regex.IsMatch(cif, @"^[0-9]+$"))
        {
            lblMessage.Text = "CIF must contain digits only.";
            return;
        }

        // Validate and standardize Emirates ID.
        string emiratesIDStandard =
            emiratesID.Replace("-", "").Replace(" ", "");

        if (!Regex.IsMatch(emiratesID, @"^784-\d{4}-\d{7}-\d$") ||
            !Regex.IsMatch(emiratesIDStandard, @"^784\d{12}$"))
        {
            lblMessage.Text =
                "Invalid Emirates ID. Expected format: 784-YYYY-NNNNNNN-N.";
            return;
        }

        // Validate date of birth.
        DateTime dob;

        if (!DateTime.TryParseExact(
            dobInput,
            "dd-MM-yyyy",
            CultureInfo.InvariantCulture,
            DateTimeStyles.None,
            out dob))
        {
            lblMessage.Text =
                "Invalid date of birth. Use DD-MM-YYYY format.";
            return;
        }

        // Standardized date of birth.
        string dobStandard =
            dob.ToString("dd MMMM yyyy",
            CultureInfo.InvariantCulture).ToUpperInvariant();

        // Validate mobile number.
        if (!Regex.IsMatch(mobile, @"^[0-9]{7,20}$"))
        {
            lblMessage.Text =
                "Mobile number must contain digits only.";
            return;
        }

        // Validate passport number.
        if (!Regex.IsMatch(passport, @"^[A-Za-z0-9]+$"))
        {
            lblMessage.Text =
                "Passport number must contain letters and digits only.";
            return;
        }

        // Validate customer name.
        if (customerName.Length > 200 ||
            Regex.IsMatch(customerName, @"[\x00-\x1F]"))
        {
            lblMessage.Text = "Invalid customer name.";
            return;
        }

        // Validate nationality.
        if (nationality.Length > 100 ||
            Regex.IsMatch(nationality, @"[\x00-\x1F]"))
        {
            lblMessage.Text = "Invalid nationality.";
            return;
        }

        // Escape single quotes for SQL string literals.
        string nameSQL = EscapeSQL(customerName);
        string cifSQL = EscapeSQL(cif);
        string eidSQL = EscapeSQL(emiratesID);
        string eidStandardSQL = EscapeSQL(emiratesIDStandard);
        string dobStandardSQL = EscapeSQL(dobStandard);
        string passportSQL = EscapeSQL(passport);
        string nationalitySQL = EscapeSQL(nationality);
        string mobileSQL = EscapeSQL(mobile);

        // Generate INSERT statement.
        StringBuilder sql = new StringBuilder();

        sql.AppendLine("INSERT INTO EDSUSER.EDS_CUSTOMER_TABLE");
        sql.AppendLine("(");
        sql.AppendLine("    CUSTOMER_FULL_NAME,");
        sql.AppendLine("    CUSTOMER_FULL_NAME_STND,");
        sql.AppendLine("    CUSTOMER_FULL_NAME_STND_ORG,");
        sql.AppendLine("    CIF_NO,");
        sql.AppendLine("    CIF_NO_STND,");
        sql.AppendLine("    EMIRATESID_NO,");
        sql.AppendLine("    EMIRATESID_NO_STND,");
        sql.AppendLine("    DATE_OF_BIRTH,");
        sql.AppendLine("    DATE_OF_BIRTH_STND,");
        sql.AppendLine("    PASSPORT_NO,");
        sql.AppendLine("    PASSPORT_NO_STND,");
        sql.AppendLine("    NATIONALITY,");
        sql.AppendLine("    NATIONALITY_STND,");
        sql.AppendLine("    MOBILE,");
        sql.AppendLine("    MOBILE_STND,");
        sql.AppendLine("    CLUSTERID");
        sql.AppendLine(")");
        sql.AppendLine("VALUES");
        sql.AppendLine("(");
        sql.AppendLine("    '" + nameSQL + "',");
        sql.AppendLine("    '" + nameSQL + "',");
        sql.AppendLine("    '" + nameSQL + "',");
        sql.AppendLine("    '" + cifSQL + "',");
        sql.AppendLine("    '" + cifSQL + "',");
        sql.AppendLine("    '" + eidSQL + "',");
        sql.AppendLine("    '" + eidStandardSQL + "',");
        sql.AppendLine("    TIMESTAMP '" +
            dob.ToString("yyyy-MM-dd", CultureInfo.InvariantCulture) +
            " 00:00:00.000000',");
        sql.AppendLine("    '" + dobStandardSQL + "',");
        sql.AppendLine("    '" + passportSQL + "',");
        sql.AppendLine("    '" + passportSQL + "',");
        sql.AppendLine("    '" + nationalitySQL + "',");
        sql.AppendLine("    '" + nationalitySQL + "',");
        sql.AppendLine("    '" + mobileSQL + "',");
        sql.AppendLine("    '" + mobileSQL + "',");
        sql.AppendLine("    " + cif);
        sql.AppendLine(");");

        txtSQL.Text = sql.ToString();

        lblMessage.ForeColor =
            System.Drawing.Color.Green;

        lblMessage.Text = "SQL script generated successfully.";
    }

    private string EscapeSQL(string value)
    {
        return value.Replace("'", "''");
    }
}
